# Check AWS CLI version
aws --version

# List EC2 instances
aws ec2 describe-instances

# List volumes
aws ec2 describe-volumes

# Create a snapshot
aws ec2 create-snapshot --volume-id <volume-id> --description "Backup snapshot"

# List snapshots
aws ec2 describe-snapshots --owner-ids self

# Create an AMI
aws ec2 create-image --instance-id <instance-id> --name "my-ami" --description "AMI for launch template"

# List AMIs
aws ec2 describe-images --owners self

# Create a Launch Template
aws ec2 create-launch-template \
  --launch-template-name my-template \
  --version-description "v1" \
  --launch-template-data '{"ImageId":"<ami-id>","InstanceType":"t3.micro"}'

# Launch instance from Launch Template
aws ec2 run-instances --launch-template LaunchTemplateName=my-template,Version=1

# Check Target Group health
aws elbv2 describe-target-health --target-group-arn <tg-arn>

# Check CloudWatch alarms
aws cloudwatch describe-alarms

# List SNS topics
aws sns list-topics
