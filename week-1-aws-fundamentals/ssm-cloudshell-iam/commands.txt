# Verify instance is registered with SSM
aws ssm describe-instance-information

# Start an SSM session (from CloudShell or local CLI)
aws ssm start-session --target <instance-id>

# List IAM roles
aws iam list-roles

# Attach IAM role to EC2 instance (if needed)
aws ec2 associate-iam-instance-profile \
  --instance-id <instance-id> \
  --iam-instance-profile Name=<profile-name>

# Check SSM agent logs (inside the instance)
sudo systemctl status amazon-ssm-agent
sudo tail -f /var/log/amazon/ssm/amazon-ssm-agent.log

# Test basic commands inside SSM session
ls -l
hostname
df -h
