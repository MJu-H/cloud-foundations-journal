# List block devices
fdisk -l

# Create a partition on the new volume
sudo fdisk /dev/xvdf

# Format the new partition
sudo mkfs -t ext4 /dev/xvdf1

# Create a mount point
sudo mkdir /data

# Mount the partition
sudo mount /dev/xvdf1 /data

# Verify the mount
df -h

# Write test files
sudo touch /data/testfile.txt

# Add to /etc/fstab for persistent mounting
# (You edited this manually using vi)
sudo vi /etc/fstab

# Example fstab entry:
# /dev/xvdf1   /data   ext4   defaults,nofail   0   2

# Reboot to test persistence
sudo reboot

# After reboot, verify the mount
df -h
fdisk -l
ls /data
